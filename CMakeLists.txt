cmake_minimum_required(VERSION 3.20)
project(ArcanaSurvivorsCpp VERSION 0.5.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ARCANA_BUILD_SERVER "Build Boost.Beast WebSocket server" ON)
option(ARCANA_BUILD_CLIENT "Build SDL2 native client" OFF)
option(ARCANA_BUILD_NET "Build native WebSocket client support" ON)
option(ARCANA_BUILD_TESTS "Build tests" ON)
option(ARCANA_BUILD_DESKTOP "Build the SDL2 native frontend for PC (same code as the Switch port)" OFF)
add_library(arcana_core src/game.cpp src/data.cpp src/profile.cpp src/native/render_queue.cpp)
target_include_directories(arcana_core PUBLIC include)
if(MSVC)
  target_compile_options(arcana_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(arcana_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(arcana_sim tools/sim.cpp)
target_link_libraries(arcana_sim PRIVATE arcana_core)
add_executable(arcana_profile tools/profile.cpp)
target_link_libraries(arcana_profile PRIVATE arcana_core)
add_executable(arcana_native_bench tools/native_bench.cpp)
target_link_libraries(arcana_native_bench PRIVATE arcana_core)
if(ARCANA_BUILD_NET)
  find_package(Threads REQUIRED)
  find_path(BOOST_INCLUDE_DIR boost/beast.hpp)
  if(BOOST_INCLUDE_DIR)
    add_library(arcana_net client/network.cpp)
    target_include_directories(arcana_net PUBLIC client ${BOOST_INCLUDE_DIR})
    target_compile_definitions(arcana_net PRIVATE BOOST_ERROR_CODE_HEADER_ONLY BOOST_SYSTEM_NO_DEPRECATED)
    target_link_libraries(arcana_net PUBLIC Threads::Threads)
  endif()
endif()
if(ARCANA_BUILD_SERVER)
  find_package(Threads REQUIRED)
  find_path(BOOST_INCLUDE_DIR boost/beast.hpp)
  if(BOOST_INCLUDE_DIR)
    add_executable(arcana_server server/main.cpp)
    target_include_directories(arcana_server PRIVATE ${BOOST_INCLUDE_DIR})
    target_compile_definitions(arcana_server PRIVATE BOOST_ERROR_CODE_HEADER_ONLY BOOST_SYSTEM_NO_DEPRECATED)
    target_link_libraries(arcana_server PRIVATE arcana_core Threads::Threads)
  else()
    message(WARNING "Boost headers not found; arcana_server disabled")
  endif()
endif()
if(ARCANA_BUILD_CLIENT)
  add_executable(arcana_client client/main.cpp)
  target_link_libraries(arcana_client PRIVATE arcana_core)

  if(EMSCRIPTEN)
    target_compile_options(arcana_client PRIVATE "-sUSE_SDL=2")
    target_link_options(arcana_client PRIVATE "-sUSE_SDL=2" "-sALLOW_MEMORY_GROWTH=1")
    set_target_properties(arcana_client PROPERTIES SUFFIX ".html")
  else()
    find_package(SDL2 REQUIRED)
    target_link_libraries(arcana_client PRIVATE SDL2::SDL2)
    if(TARGET arcana_net)
      target_link_libraries(arcana_client PRIVATE arcana_net)
      target_compile_definitions(arcana_client PRIVATE ARCANA_HAS_NET=1)
    endif()
  endif()
endif()
if(ARCANA_BUILD_DESKTOP)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2 SDL2_image SDL2_ttf)
  add_library(arcana_frontend platforms/sdl/batch_renderer.cpp platforms/sdl/frontend.cpp platforms/sdl/animator.cpp platforms/sdl/world_renderer.cpp platforms/sdl/audio.cpp)
  target_include_directories(arcana_frontend PUBLIC platforms/sdl)
  target_link_libraries(arcana_frontend PUBLIC arcana_core PkgConfig::SDL2)
  add_executable(arcana_desktop platforms/desktop/main.cpp)
  target_link_libraries(arcana_desktop PRIVATE arcana_frontend)
endif()
if(ARCANA_BUILD_TESTS)
  enable_testing()
  add_executable(arcana_core_tests tests/core_tests.cpp)
  target_link_libraries(arcana_core_tests PRIVATE arcana_core)
  add_test(NAME core COMMAND arcana_core_tests)
  add_executable(arcana_native_hotpath_tests tests/native_hotpath_tests.cpp)
  target_link_libraries(arcana_native_hotpath_tests PRIVATE arcana_core)
  add_test(NAME native_hotpath COMMAND arcana_native_hotpath_tests)
  if(TARGET arcana_frontend)
    add_executable(arcana_frontend_tests tests/frontend_tests.cpp)
    target_link_libraries(arcana_frontend_tests PRIVATE arcana_frontend)
    add_test(NAME frontend COMMAND arcana_frontend_tests)
  endif()
  if(TARGET arcana_net)
    add_executable(arcana_net_compile tests/network_compile.cpp)
    target_link_libraries(arcana_net_compile PRIVATE arcana_net)
    add_test(NAME net_compile COMMAND arcana_net_compile)
  endif()
endif()
